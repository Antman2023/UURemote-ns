cmake_minimum_required(VERSION 3.21)
project(UURemote-ns VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
option(UU_SWITCH "Build Nintendo Switch homebrew" OFF)
option(UU_CORE_ONLY "Only compile portable protocol tests" OFF)
option(UU_RTC_TESTS "Build native WebRTC loopback tests instead of the UI" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="uu_mbedtls_config.h")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/include")
add_subdirectory(external/mbedtls EXCLUDE_FROM_ALL)
add_library(MbedTLS::MbedTLS INTERFACE IMPORTED GLOBAL)
set_target_properties(MbedTLS::MbedTLS PROPERTIES INTERFACE_LINK_LIBRARIES mbedtls INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/external/mbedtls/include")
add_library(MbedTLS::MbedCrypto ALIAS mbedcrypto)
add_library(MbedTLS::MbedX509 ALIAS mbedx509)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
add_subdirectory(external/json EXCLUDE_FROM_ALL)
add_library(uu_protocol source/protocol.cpp)
target_link_libraries(uu_protocol PUBLIC nlohmann_json::nlohmann_json mbedcrypto)
target_include_directories(uu_protocol PUBLIC include)
if(UU_CORE_ONLY)
    enable_testing()
    add_executable(protocol_test tests/protocol_test.cpp)
    target_link_libraries(protocol_test PRIVATE uu_protocol)
    add_test(NAME protocol COMMAND protocol_test)
    return()
endif()

set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(USE_SYSTEM_JSON ON CACHE BOOL "" FORCE)
set(ENABLE_MBEDTLS OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENSSL OFF CACHE BOOL "" FORCE)
set(ENABLE_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(LIBSRTP_TEST_APPS OFF CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(INSTALL_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
set(nlohmann_json_DIR "${CMAKE_BINARY_DIR}/external/json" CACHE PATH "" FORCE)
add_subdirectory(external/libdatachannel EXCLUDE_FROM_ALL)
if(UU_SWITCH)
    target_include_directories(usrsctp PRIVATE "${CMAKE_SOURCE_DIR}/include/switch_compat")
    target_compile_options(srtp2 PRIVATE -include arpa/inet.h)
    target_compile_options(datachannel-static PRIVATE -include arpa/inet.h)
    target_compile_options(datachannel PRIVATE -include arpa/inet.h)
    target_include_directories(juice PRIVATE "${CMAKE_SOURCE_DIR}/include/switch_compat")
    target_include_directories(juice-static PRIVATE "${CMAKE_SOURCE_DIR}/include/switch_compat")
endif()
if(UU_RTC_TESTS)
    enable_testing()
    add_executable(rtc_test tests/rtc_test.cpp)
    target_link_libraries(rtc_test PRIVATE datachannel-static)
    add_test(NAME rtc_loopback COMMAND rtc_test)
    set_tests_properties(rtc_loopback PROPERTIES TIMEOUT 35)
    return()
endif()
add_library(qrcodegen external/qrcodegen/cpp/qrcodegen.cpp)
target_include_directories(qrcodegen PUBLIC external/qrcodegen/cpp)
add_executable(UURemote-ns source/main.cpp source/client.cpp source/media.cpp source/transport.cpp)
target_link_libraries(UURemote-ns PRIVATE uu_protocol datachannel-static qrcodegen)
target_compile_options(UURemote-ns PRIVATE -Wall -Wextra)
if(UU_SWITCH)
    target_sources(UURemote-ns PRIVATE source/platform_switch.cpp)
    file(TO_CMAKE_PATH "$ENV{DEVKITPRO}" DEVKITPRO)
    target_include_directories(UURemote-ns PRIVATE "${DEVKITPRO}/portlibs/switch/include/SDL2" "${DEVKITPRO}/portlibs/switch/include/opus")
    target_link_libraries(UURemote-ns PRIVATE -Wl,--start-group SDL2_ttf SDL2 freetype harfbuzz png16 EGL glapi drm_nouveau avcodec avutil swscale swresample dav1d opus curl z bz2 nx pthread m -Wl,--end-group)
    set_target_properties(UURemote-ns PROPERTIES SUFFIX ".elf")
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/UURemote-ns.nro"
        COMMAND "${DEVKITPRO}/tools/bin/nacptool" --create "UU Remote NS" "UURemote contributors" "${PROJECT_VERSION}" "${CMAKE_BINARY_DIR}/UURemote-ns.nacp"
        COMMAND "${DEVKITPRO}/tools/bin/elf2nro" "$<TARGET_FILE:UURemote-ns>" "${CMAKE_BINARY_DIR}/UURemote-ns.nro" "--nacp=${CMAKE_BINARY_DIR}/UURemote-ns.nacp" "--icon=${DEVKITPRO}/libnx/default_icon.jpg" "--romfsdir=${CMAKE_SOURCE_DIR}/romfs"
        DEPENDS UURemote-ns "${CMAKE_SOURCE_DIR}/romfs/signing.key" VERBATIM)
    add_custom_target(nro ALL DEPENDS "${CMAKE_BINARY_DIR}/UURemote-ns.nro")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf libavcodec libavutil libswscale opus libcurl)
    target_link_libraries(UURemote-ns PRIVATE PkgConfig::NATIVE)
endif()
